    81→function ValueSelect({
    82→  kind,
    83→  value,
    84→  onChange,
    85→  options: optionsOverride,
    86→}: {
    87→  kind: ValueKind;
    88→  value: string;
    89→  onChange: (v: string) => void;
    90→  /** Explicit option list — takes precedence over the workspace-derived ones. */
    91→  options?: { value: string; label: string }[];
    92→}) {
    93→  const workspace = usePortalStore((s) => s.workspace);
    94→  const options: { value: string; label: string }[] = optionsOverride ?? (() => {
    95→    if (!workspace) return [];
    96→    switch (kind) {
    97→      case "status":
    98→        return [...workspace.statuses].sort((a, b) => a.order - b.order).map((s) => ({ value: s.name, label: s.name }));
    99→      case "type":
   100→        return [...workspace.issueTypes].sort((a, b) => a.order - b.order).map((t) => ({ value: t.name, label: t.name }));
   101→      case "priority":
   102→        return [...workspace.priorities].sort((a, b) => a.order - b.order).map((p) => ({ value: p.name, label: p.name }));
   103→      case "label":
   104→        return workspace.labels.map((l) => ({ value: l.name, label: l.name }));
   105→      case "member":
   106→        return workspace.members.map((m) => ({ value: m.id, label: m.name }));
   107→      default:
   108→        return [];
   109→    }
   110→  })();
   111→
   112→  if (kind === "none") return <span className="text-xs italic text-muted-foreground">no value needed</span>;
   113→  if (kind === "free") {
   114→    return (
   115→      <Input
   116→        value={value}
   117→        onChange={(e) => onChange(e.target.value)}
   118→        placeholder="value"
   119→        className="h-8 flex-1 text-xs"
   120→      />
   121→    );
   122→  }
   123→  return (
   124→    <Select value={value} onValueChange={onChange}>
   125→      <SelectTrigger className="h-8 min-w-32 flex-1 text-xs" aria-label="Value">
   126→        <SelectValue placeholder="Pick value" />
   127→      </SelectTrigger>
   128→      <SelectContent>
   129→        {options.map((o) => (
   130→          <SelectItem key={o.value} value={o.value} className="text-xs">
   131→            {o.label}
   132→          </SelectItem>
   133→        ))}
   134→      </SelectContent>
   135→    </Select>
   136→  );
   137→}

   139→function RuleDialog({
   140→  open,
   141→  onOpenChange,
   142→  editing,
   143→  onSaved,
   144→}: {
   145→  open: boolean;
   146→  onOpenChange: (v: boolean) => void;
   147→  editing: AutomationRuleDTO | null;
   148→  onSaved: () => void;
   149→}) {
   150→  const [name, setName] = useState("");
   151→  const [trigger, setTrigger] = useState("issue.status_changed");
   152→  const [conditions, setConditions] = useState<AutomationConditionDTO[]>([]);
   153→  const [actions, setActions] = useState<AutomationActionDTO[]>([{ type: "notify_assignee" }]);
   154→  const [busy, setBusy] = useState(false);
   155→  const [testing, setTesting] = useState(false);
   156→  const [testResult, setTestResult] = useState<
   157→    | {
   158→        scanned: number;
   159→        matchCount: number;
   160→        truncated: boolean;
   161→        matches: {
   162→          id: string; key: string; summary: string;
   163→          status: { name: string; color: string };
   164→          type: { name: string };
   165→          assignee: { id: string; name: string } | null;
   166→          wouldApply: string[];
   167→        }[];
   168→      }
   169→    | null
   170→  >(null);
   171→
   172→  useEffect(() => {
   173→    if (!open) return;
   174→    if (editing) {
   175→      setName(editing.name);
   176→      setTrigger(editing.trigger);
   177→      setConditions(editing.conditions.map((c) => ({ ...c })));
   178→      setActions(editing.actions.map((a) => ({ ...a })));
   179→    } else {
   180→      setName("");
   181→      setTrigger("issue.status_changed");
   182→      setConditions([{ field: "type", operator: "equals", value: "Bug" }]);
   183→      setActions([{ type: "notify_assignee" }]);
   184→    }
   185→  }, [open, editing]);
   186→
   187→  function conditionFieldKind(field: string): "list" | "free" {
   188→    return field === "summary" ? "free" : "list";
   189→  }
   190→
   191→  function conditionValueOptions(field: string): { value: string; label: string }[] {
   192→    const workspace = usePortalStore.getState().workspace;
   193→    if (!workspace) return [];
   194→    switch (field) {
   195→      case "type":
   196→        return [...workspace.issueTypes].sort((a, b) => a.order - b.order).map((t) => ({ value: t.name, label: t.name }));
   197→      case "status":
   198→        return [...workspace.statuses].sort((a, b) => a.order - b.order).map((s) => ({ value: s.name, label: s.name }));
   199→      case "priority":
   200→        return [...workspace.priorities].sort((a, b) => a.order - b.order).map((p) => ({ value: p.name, label: p.name }));
   201→      case "label":
   202→        return workspace.labels.map((l) => ({ value: l.name, label: l.name }));
   203→      case "assignee":
   204→        return [
   205→          { value: "me", label: "Me" },
   206→          { value: "none", label: "Unassigned" },
   207→        ];
   208→      case "points":
   209→        return ["1", "2", "3", "5", "8", "13"].map((n) => ({ value: n, label: n }));
   210→      default:
   211→        return [];
   212→    }
   213→  }
   214→
   215→  async function runTest() {
   216→    setTesting(true);
   217→    try {
   218→      const res = await api2.testAutomation({
   219→        conditions: conditions.filter((c) => c.value || conditionFieldKind(c.field) === "free"),
   220→        actions: actions.map((a) => ({ type: a.type, value: a.value || undefined })),
   221→      });
   222→      setTestResult(res);
   223→    } catch (err) {
   224→      toast.error(err instanceof Error ? err.message : "Test run failed");
   225→    } finally {
   226→      setTesting(false);
   227→    }
   228→  }
   229→
   230→  async function submit() {
   231→    if (!name.trim()) return toast.error("Give the rule a name");
   232→    if (!actions.length) return toast.error("Add at least one action");
   233→    setBusy(true);
   234→    try {
   235→      const payload = {
   236→        name: name.trim(),
   237→        trigger,
   238→        conditions: conditions.filter((c) => c.value || conditionFieldKind(c.field) === "free"),
   239→        actions: actions.map((a) => ({ type: a.type, value: a.value || undefined })),
   240→      };
   241→      if (editing) {
   242→        await api2.patchAutomation(editing.id, payload);
   243→        toast.success("Rule updated");
   244→      } else {
   245→        await api2.createAutomation(payload);
   246→        toast.success("Automation created", { description: "It runs on the next matching event." });
   247→      }
   248→      onOpenChange(false);
   249→      onSaved();
   250→    } catch (err) {
   251→      toast.error(err instanceof Error ? err.message : "Failed to save rule");
   252→    } finally {
   253→      setBusy(false);
   254→    }
   255→  }
   256→
   257→  return (
   258→    <Dialog open={open} onOpenChange={onOpenChange}>
   259→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   260→        <DialogHeader>
   261→          <DialogTitle className="flex items-center gap-2">
   262→            <Zap className="size-4 text-amber-600" aria-hidden />
   263→            {editing ? "Edit automation" : "New automation"}
   264→          </DialogTitle>
   265→          <DialogDescription>When <em>trigger</em>, if <em>conditions</em>, then <em>actions</em>.</DialogDescription>
   266→        </DialogHeader>
   267→
   268→        <div className="space-y-4">
   269→          <div className="space-y-1.5">
   270→            <label className="text-xs font-medium text-foreground" htmlFor="rule-name">Rule name</label>
   271→            <Input
   272→              id="rule-name"
   273→              value={name}
   274→              onChange={(e) => setName(e.target.value)}
   275→              placeholder="e.g. Auto-notify on critical bugs"
   276→            />
   277→          </div>
   278→
   279→          {/* WHEN */}
   280→          <div className="rounded-lg border border-border p-3">
   281→            <div className="mb-2 flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-wide text-amber-700">
   282→              <Play className="size-3" aria-hidden /> When
   283→            </div>
   284→            <Select value={trigger} onValueChange={setTrigger}>
   285→              <SelectTrigger className="h-8 text-xs" aria-label="Trigger">
   286→                <SelectValue />
   287→              </SelectTrigger>
   288→              <SelectContent>
   289→                {TRIGGERS.map((t) => (
   290→                  <SelectItem key={t.value} value={t.value} className="text-xs">
   291→                    {t.label}
   292→                  </SelectItem>
   293→                ))}
   294→              </SelectContent>
   295→            </Select>
   296→          </div>
   297→
   298→          {/* IF */}
   299→          <div className="rounded-lg border border-border p-3">
   300→            <div className="mb-2 flex items-center justify-between">
   301→              <div className="flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-wide text-violet-700 dark:text-violet-300">
   302→                <Filter className="size-3" aria-hidden /> If (all match)
   303→              </div>
   304→              <button
   305→                type="button"
   306→                className="flex items-center gap-1 rounded px-1.5 py-0.5 text-[11px] font-medium text-muted-foreground hover:bg-muted hover:text-foreground"
   307→                onClick={() =>
   308→                  setConditions((c) => [...c, { field: "type", operator: "equals", value: "Bug" }])
   309→                }
   310→              >
   311→                <Plus className="size-3" aria-hidden /> Condition
   312→              </button>
   313→            </div>
   314→            {conditions.length === 0 ? (
   315→              <p className="py-1 text-xs italic text-muted-foreground">No conditions — always runs.</p>
   316→            ) : (
   317→              <div className="space-y-2">
   318→                {conditions.map((cond, i) => (
   319→                  <div key={i} className="flex items-center gap-1.5">
   320→                    <Select
   321→                      value={cond.field}
   322→                      onValueChange={(v) =>
   323→                        setConditions((c) => c.map((x, j) => (j === i ? { ...x, field: v, value: "" } : x)))
   324→                      }
   325→                    >
   326→                      <SelectTrigger className="h-8 w-28 shrink-0 text-xs" aria-label="Field">
   327→                        <SelectValue />
   328→                      </SelectTrigger>
   329→                      <SelectContent>
   330→                        {CONDITION_FIELDS.map((f) => (
   331→                          <SelectItem key={f.value} value={f.value} className="text-xs">
   332→                            {f.label}
   333→                          </SelectItem>
   334→                        ))}
   335→                      </SelectContent>
   336→                    </Select>
   337→                    <Select
   338→                      value={cond.operator}
   339→                      onValueChange={(v) =>
   340→                        setConditions((c) => c.map((x, j) => (j === i ? { ...x, operator: v as AutomationConditionDTO["operator"] } : x)))
   341→                      }
   342→                    >
   343→                      <SelectTrigger className="h-8 w-32 shrink-0 text-xs" aria-label="Operator">
   344→                        <SelectValue />
   345→                      </SelectTrigger>
   346→                      <SelectContent>
   347→                        {OPERATORS.map((o) => (
   348→                          <SelectItem key={o.value} value={o.value} className="text-xs">
   349→                            {o.label}
   350→                          </SelectItem>
   351→                        ))}
   352→                      </SelectContent>
   353→                    </Select>
   354→                    {conditionFieldKind(cond.field) === "free" ? (
   355→                      <Input
   356→                        value={cond.value}
   357→                        onChange={(e) =>
   358→                          setConditions((c) => c.map((x, j) => (j === i ? { ...x, value: e.target.value } : x)))
   359→                        }
   360→                        placeholder="text…"
   361→                        className="h-8 flex-1 text-xs"
   362→                      />
   363→                    ) : (
   364→                      <ValueSelect
   365→                        kind="list"
   366→                        options={conditionValueOptions(cond.field)}
   367→                        value={cond.value}
   368→                        onChange={(v) => setConditions((c) => c.map((x, j) => (j === i ? { ...x, value: v } : x)))}
   369→                      />
   370→                    )}
   371→                    <button
   372→                      type="button"
   373→                      aria-label={`Remove condition ${i + 1}`}
   374→                      className="rounded p-1 text-muted-foreground/60 hover:bg-rose-500/10 hover:text-rose-600"
   375→                      onClick={() => setConditions((c) => c.filter((_, j) => j !== i))}
   376→                    >
   377→                      <Trash2 className="size-3.5" aria-hidden />
   378→                    </button>
   379→                  </div>
   380→                ))}
   381→              </div>
   382→            )}
   383→          </div>
   384→
   385→          {/* THEN */}
   386→          <div className="rounded-lg border border-border p-3">
   387→            <div className="mb-2 flex items-center justify-between">
   388→              <div className="flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-wide text-emerald-700">
   389→                <GitBranch className="size-3" aria-hidden /> Then
   390→              </div>
   391→            </div>
   392→            <div className="space-y-2">
   393→              {actions.map((action, i) => {
   394→                const meta = ACTIONS.find((a) => a.value === action.type);
   395→                return (
   396→                  <div key={i} className="flex items-center gap-1.5">
   397→                    <Select
   398→                      value={action.type}
   399→                      onValueChange={(v) =>
   400→                        setActions((a) => a.map((x, j) => (j === i ? { type: v, value: undefined } : x)))
   401→                      }
   402→                    >
   403→                      <SelectTrigger className="h-8 min-w-40 flex-1 text-xs" aria-label="Action">
   404→                        <SelectValue />
   405→                      </SelectTrigger>
   406→                      <SelectContent>
   407→                        {ACTIONS.map((a) => (
   408→                          <SelectItem key={a.value} value={a.value} className="text-xs">
   409→                            {a.label}
   410→                          </SelectItem>
   411→                        ))}
   412→                      </SelectContent>
   413→                    </Select>
   414→                    {meta?.valueKind && meta.valueKind !== "none" && (
   415→                      <ValueSelect
   416→                        kind={meta.valueKind}
   417→                        value={action.value ?? ""}
   418→                        onChange={(v) => setActions((a) => a.map((x, j) => (j === i ? { ...x, value: v } : x)))}
   419→                      />
   420→                    )}
   421→                    <button
   422→                      type="button"
   423→                      aria-label={`Remove action ${i + 1}`}
   424→                      className="rounded p-1 text-muted-foreground/60 hover:bg-rose-500/10 hover:text-rose-600"
   425→                      onClick={() => setActions((a) => a.filter((_, j) => j !== i))}
   426→                    >
   427→                      <Trash2 className="size-3.5" aria-hidden />
   428→                    </button>
   429→                  </div>
   430→                );
   431→              })}
   432→              <button
   433→                type="button"
   434→                className="flex items-center gap-1 rounded px-1.5 py-0.5 text-[11px] font-medium text-muted-foreground hover:bg-muted hover:text-foreground"
   435→                onClick={() => setActions((a) => [...a, { type: "add_label", value: "" }])}
   436→              >
   437→                <Plus className="size-3" aria-hidden /> Action
   438→              </button>
   439→            </div>
   440→          </div>
   441→        </div>
   442→
   443→        {/* Test-run preview */}
   444→        {(testing || testResult) && (
   445→          <div
   446→            className="rounded-lg border border-amber-500/30 bg-amber-500/[0.04] p-3"
   447→            aria-live="polite"
   448→          >
   449→            <div className="mb-2 flex items-center gap-1.5 text-[11px] font-bold uppercase tracking-wide text-amber-700 dark:text-amber-400">
   450→              <FlaskConical className="size-3" aria-hidden /> Dry run — no changes applied
   451→            </div>
   452→            {testing ? (
   453→              <div className="flex items-center gap-2 py-2 text-xs text-muted-foreground">
   454→                <Loader2 className="size-3.5 animate-spin" aria-hidden /> Matching against the 50 most recently updated issues…
   455→              </div>
   456→            ) : testResult ? (
   457→              testResult.matchCount === 0 ? (
   458→                <p className="py-1 text-xs text-muted-foreground">
   459→                  No issues out of the last {testResult.scanned} updated would match these conditions.
   460→                </p>
   461→              ) : (
   462→                <div className="space-y-2">
   463→                  <p className="text-xs text-muted-foreground">
   464→                    <span className="font-semibold text-foreground">{testResult.matchCount}</span> of{" "}
   465→                    {testResult.scanned} recent issues would match
   466→                    {testResult.truncated ? " (showing first 10)" : ""}:
   467→                  </p>
   468→                  <ul className="max-h-44 space-y-1 overflow-y-auto pr-1 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-stone-300 [&::-webkit-scrollbar]:w-1.5">
   469→                    {testResult.matches.map((m) => (
   470→                      <li
   471→                        key={m.id}
   472→                        className="rounded-md border border-border/70 bg-card px-2.5 py-2"
   473→                      >
   474→                        <div className="flex items-center gap-2">
   475→                          <span
   476→                            className="size-2 shrink-0 rounded-full"
   477→                            style={{ backgroundColor: m.status.color }}
   478→                            aria-hidden
   479→                          />
   480→                          <span className="shrink-0 font-mono text-[11px] font-semibold text-muted-foreground">{m.key}</span>
   481→                          <span className="min-w-0 flex-1 truncate text-xs text-foreground">{m.summary}</span>
   482→                        </div>
   483→                        {m.wouldApply.length > 0 && (
   484→                          <div className="mt-1.5 flex flex-wrap gap-1 pl-4">
   485→                            {m.wouldApply.map((w, wi) => (
   486→                              <span
   487→                                key={wi}
   488→                                className="rounded bg-emerald-500/10 px-1.5 py-px text-[10px] font-medium text-emerald-700 dark:text-emerald-400"
   489→                              >
   490→                                {w}
   491→                              </span>
   492→                            ))}
   493→                          </div>
   494→                        )}
   495→                      </li>
   496→                    ))}
   497→                  </ul>
   498→                </div>
   499→              )
   500→            ) : null}
   501→          </div>
   502→        )}
   503→
   504→        <DialogFooter className="flex-row items-center justify-between gap-2 sm:justify-between">
   505→          <Button
   506→            type="button"
   507→            variant="outline"
   508→            className="gap-1.5"
   509→            onClick={() => void runTest()}
   510→            disabled={testing || busy}
   511→            title="Preview which recent issues this rule would match — nothing is changed"
   512→          >
   513→            {testing ? (
   514→              <Loader2 className="size-3.5 animate-spin" aria-hidden />
   515→            ) : (
   516→              <FlaskConical className="size-3.5" aria-hidden />
   517→            )}
   518→            Test run
   519→          </Button>
   520→          <span className="flex items-center gap-2">
   521→            <Button variant="outline" onClick={() => onOpenChange(false)}>
   522→              Cancel
   523→            </Button>
   524→            <Button
   525→              className="gap-1.5 bg-amber-600 text-white hover:bg-amber-700"
   526→              onClick={() => void submit()}
   527→              disabled={busy}
   528→            >
   529→              <Zap className="size-3.5" aria-hidden /> {editing ? "Save changes" : "Create rule"}
   530→            </Button>
   531→          </span>
   532→        </DialogFooter>
   533→      </DialogContent>
   534→    </Dialog>

   538→function triggerLabel(t: string) {
   539→  return TRIGGERS.find((x) => x.value === t)?.label ?? t;
   540→}

   542→function describeCondition(c: AutomationConditionDTO) {
   543→  const field = CONDITION_FIELDS.find((f) => f.value === c.field)?.label ?? c.field;
   544→  const op = OPERATORS.find((o) => o.value === c.operator)?.label ?? c.operator;
   545→  return `${field} ${op} ${c.value || "…"}`;
   546→}
   547→
   548→function describeAction(a: AutomationActionDTO) {
   549→  const meta = ACTIONS.find((x) => x.value === a.type);
   550→  const label = meta?.label ?? a.type;
   551→  return a.value ? `${label}` : label;
   552→}
