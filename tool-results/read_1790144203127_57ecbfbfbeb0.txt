    46→// ─── Layout constants (deterministic graph geometry) ────────────

    48→const NODE_W = 208;
    49→const NODE_H = 62;
    50→const COL_GAP = 96;
    51→const ROW_GAP = 18;
    52→const GRAPH_TOP = 56; // space for column headers
    53→const GRAPH_PAD = 24;
    54→

    66→interface NodePos {
    67→  status: WorkflowStatusDTO;
    68→  x: number;
    69→  y: number;
    70→  col: number;
    71→}
    72→
    73→function layout(statuses: WorkflowStatusDTO[]): { nodes: NodePos[]; width: number; height: number } {
    74→  const byCat: Record<StatusCategory, WorkflowStatusDTO[]> = { TODO: [], IN_PROGRESS: [], DONE: [] };
    75→  for (const s of statuses) byCat[s.category]?.push(s);
    76→  (Object.keys(byCat) as StatusCategory[]).forEach((k) =>
    77→    byCat[k].sort((a, b) => a.order - b.order)
    78→  );
    79→
    80→  const nodes: NodePos[] = [];
    81→  let maxRows = 0;
    82→  CATEGORIES.forEach((cat, col) => {
    83→    byCat[cat.key].forEach((status, row) => {
    84→      nodes.push({
    85→        status,
    86→        col,
    87→        x: GRAPH_PAD + col * (NODE_W + COL_GAP),
    88→        y: GRAPH_TOP + row * (NODE_H + ROW_GAP),
    89→      });
    90→    });
    91→    maxRows = Math.max(maxRows, byCat[cat.key].length);
    92→  });
    93→
    94→  const width = GRAPH_PAD * 2 + 2 * NODE_W + 2 * COL_GAP;
    95→  const height = GRAPH_TOP + Math.max(maxRows, 2) * (NODE_H + ROW_GAP) + ROW_GAP;
    96→  return { nodes, width, height };
    97→}
    98→
    99→/** Cubic bezier path between two node edges. */
   100→function edgePath(a: NodePos, b: NodePos): string {
   101→  const x1 = a.x + NODE_W;
   102→  const y1 = a.y + NODE_H / 2;
   103→  const x2 = b.x;
   104→  const y2 = b.y + NODE_H / 2;
   105→
   106→  if (b.col > a.col) {
   107→    const dx = Math.max(36, (x2 - x1) * 0.5);
   108→    return `M ${x1} ${y1} C ${x1 + dx} ${y1}, ${x2 - dx} ${y2}, ${x2} ${y2}`;
   109→  }
   110→  if (b.col === a.col) {
   111→    // same column: loop out to the right
   112→    const bulge = 46;
   113→    const cx = x1 + bulge;
   114→    return `M ${x1} ${y1} C ${cx} ${y1}, ${cx} ${y2}, ${x1} ${y2}`;
   115→  }
   116→  // backward edge: loop out to the right of the source then back left
   117→  const bulge = 56 + (a.col - b.col) * 44;
   118→  const cy = Math.max(y1, y2) + 34;
   119→  return `M ${x1} ${y1} C ${x1 + bulge} ${y1}, ${x2 - bulge} ${cy}, ${x2} ${y2 - 8}`;
   120→}

   497→                    <Button
   498→                      variant="ghost"

   506→                  )}

   513→
   514→      {/* Dialogs */}
   515→      <StatusDialog

   527→      )}
   528→    </div>
   529→  );

   659→function EditStatusDialog({
   660→  status,
   661→  statuses,
   662→  onClose,
   663→  onSaved,
   664→}: {
   665→  status: WorkflowStatusDTO;
   666→  statuses: WorkflowStatusDTO[];
   667→  onClose: () => void;
   668→  onSaved: () => Promise<void>;
   669→}) {
   670→  const [name, setName] = useState(status.name);
   671→  const [category, setCategory] = useState<StatusCategory>(status.category);
   672→  const [color, setColor] = useState(status.color);
   673→  const [isInitial, setIsInitial] = useState(status.isInitial);
   674→  const [saving, setSaving] = useState(false);
   675→  const [deleteOpen, setDeleteOpen] = useState(false);
   676→  const [moveTo, setMoveTo] = useState<string>("__auto__");
   677→  const [deleting, setDeleting] = useState(false);
   678→
   679→  const others = statuses.filter((s) => s.id !== status.id);
   680→  const dirty =
   681→    name !== status.name || category !== status.category || color !== status.color || isInitial !== status.isInitial;
   682→
   683→  async function save() {
   684→    if (!name.trim()) {
   685→      toast.error("Name cannot be empty");
   686→      return;
   687→    }
   688→    setSaving(true);
   689→    try {
   690→      await api3.patchStatus(status.id, {
   691→        name: name.trim(),
   692→        category,
   693→        color,
   694→        isInitial,
   695→      });
   696→      toast.success("Status updated");
   697→      onClose();
   698→      await onSaved();
   699→    } catch (err) {
   700→      toast.error(err instanceof Error ? err.message : "Failed to update status");
   701→    } finally {
   702→      setSaving(false);
   703→    }
   704→  }
   705→
   706→  async function doDelete() {
   707→    setDeleting(true);
   708→    try {
   709→      await api3.deleteStatus(status.id, moveTo === "__auto__" ? undefined : moveTo);
   710→      toast.success(`Status "${status.name}" deleted`);
   711→      setDeleteOpen(false);
   712→      onClose();
   713→      await onSaved();
   714→    } catch (err) {
   715→      toast.error(err instanceof Error ? err.message : "Failed to delete status");
   716→    } finally {
   717→      setDeleting(false);
   718→    }
   719→  }
   720→
   721→  return (
   722→    <>
   723→      <Dialog open onOpenChange={(o) => !o && onClose()}>
   724→        <DialogContent className="sm:max-w-md">
   725→          <DialogHeader>

   727→            <DialogDescription>
   728→              {status.issueCount} issue{status.issueCount === 1 ? "" : "s"} currently in this status.
   729→            </DialogDescription>
   730→          </DialogHeader>
   731→          <div className="space-y-4">
   732→            <div className="space-y-1.5">
   733→              <Label htmlFor="edit-status-name">Name</Label>
   734→              <Input
   735→                id="edit-status-name"
   736→                value={name}
   737→                onChange={(e) => setName(e.target.value)}
   738→                maxLength={40}
   739→              />
   740→            </div>
   741→            <div className="grid grid-cols-2 gap-3">
   742→              <div className="space-y-1.5">
   743→                <Label>Category</Label>
   744→                <Select value={category} onValueChange={(v) => setCategory(v as StatusCategory)}>
   745→                  <SelectTrigger>
   746→                    <SelectValue />
   747→                  </SelectTrigger>
   748→                  <SelectContent>
   749→                    {CATEGORIES.map((c) => (
   750→                      <SelectItem key={c.key} value={c.key}>
   751→                        {c.label}
   752→                      </SelectItem>
   753→                    ))}
   754→                  </SelectContent>
   755→                </Select>
   756→              </div>
   757→              <div className="space-y-1.5">
   758→                <Label>Color</Label>
   759→                <div className="flex flex-wrap gap-1 pt-1.5">
   760→                  {STATUS_COLORS.map((c) => (
   761→                    <button
   762→                      key={c}
   763→                      type="button"
   764→                      aria-label={`Color ${c}`}
   765→                      onClick={() => setColor(c)}
   766→                      className={cn(
   767→                        "size-5 rounded-full border-2 transition-transform hover:scale-110",
   768→                        color === c ? "border-foreground scale-110" : "border-transparent"
   769→                      )}
   770→                      style={{ backgroundColor: c }}
   771→                    />
   772→                  ))}
   773→                </div>
   774→              </div>
   775→            </div>
   776→            <div className="flex items-center justify-between rounded-lg border p-3">
   777→              <div>
   778→                <div className="flex items-center gap-1.5 text-sm font-medium">
   779→                  <Flag className="size-3.5 text-amber-600" />
   780→                  Initial status
   781→                </div>
